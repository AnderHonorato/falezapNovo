### 58. Catálogo de produtos com carrinho de compras
**Descrição:** Usuário adiciona produtos ao carrinho e finaliza pedido diretamente no WhatsApp.

**É possível implementar via API:** Sim (parcial)

**Como implementar:**
- Enviar mensagens de catálogo (já explicado anteriormente)
- Receber pedidos via webhook (type: `order`)
- Implementar lógica de carrinho no backend
- Processar pagamento separadamente

**Exemplo prático:**
```javascript
// Sistema de carrinho (gerenciado no backend)
const carrinhos = new Map();

// Adicionar produto ao carrinho
async function adicionarAoCarrinho(numeroCliente, produtoId, quantidade) {
  if (!carrinhos.has(numeroCliente)) {
    carrinhos.set(numeroCliente, []);
  }
  
  const carrinho = carrinhos.get(numeroCliente);
  const produto = await buscarProduto(produtoId);
  
  carrinho.push({
    produtoId: produtoId,
    nome: produto.nome,
    preco: produto.preco,
    quantidade: quantidade,
    subtotal: produto.preco * quantidade
  });
  
  // Confirmar adição
  await enviarMensagem(
    numeroCliente,
    `✅ ${produto.nome} (${quantidade}x) adicionado ao carrinho!\n\n` +
    `Subtotal: R$ ${(produto.preco * quantidade).toFixed(2)}\n\n` +
    `Digite "carrinho" para ver todos os itens ou "finalizar" para concluir o pedido.`
  );
}

// Ver carrinho
async function exibirCarrinho(numeroCliente) {
  const carrinho = carrinhos.get(numeroCliente) || [];
  
  if (carrinho.length === 0) {
    await enviarMensagem(numeroCliente, '🛒 Seu carrinho está vazio.');
    return;
  }
  
  const total = carrinho.reduce((sum, item) => sum + item.subtotal, 0);
  
  let mensagem = '🛒 *Seu Carrinho*\n\n';
  carrinho.forEach((item, index) => {
    mensagem += `${index + 1}. ${item.nome}\n`;
    mensagem += `   ${item.quantidade}x R$ ${item.preco.toFixed(2)} = R$ ${item.subtotal.toFixed(2)}\n\n`;
  });
  
  mensagem += `*Total: R$ ${total.toFixed(2)}*\n\n`;
  mensagem += 'Digite "finalizar" para concluir o pedido.';
  
  await enviarMensagem(numeroCliente, mensagem);
}

// Finalizar pedido
async function finalizarPedido(numeroCliente) {
  const carrinho = carrinhos.get(numeroCliente);
  
  if (!carrinho || carrinho.length === 0) {
    await enviarMensagem(numeroCliente, '❌ Carrinho vazio.');
    return;
  }
  
  const total = carrinho.reduce((sum, item) => sum + item.subtotal, 0);
  
  // Salvar pedido
  const pedido = await db.pedidos.insert({
    cliente: numeroCliente,
    itens: carrinho,
    total: total,
    status: 'aguardando_pagamento',
    criadoEm: new Date()
  });
  
  // Limpar carrinho
  carrinhos.delete(numeroCliente);
  
  // Enviar confirmação com opções de pagamento
  await enviarMensagemComBotoes(numeroCliente, {
    texto: `✅ Pedido #${pedido._id} criado!\n\n` +
           `Total: R$ ${total.toFixed(2)}\n\n` +
           `Escolha a forma de pagamento:`,
    botoes: [
      {id: 'pix', titulo: 'PIX'},
      {id: 'cartao', titulo: 'Cartão'},
      {id: 'boleto', titulo: 'Boleto'}
    ]
  });
}

// Processar escolha de pagamento
async function processarPagamento(numeroCliente, pedidoId, formaPagamento) {
  const pedido = await db.pedidos.findOne({_id: pedidoId});
  
  switch(formaPagamento) {
    case 'pix':
      const pixCode = gerarCodigoPix(pedido.total, pedidoId);
      await enviarMensagem(
        numeroCliente,
        `💰 *Pagamento via PIX*\n\n` +
        `Valor: R$ ${pedido.total.toFixed(2)}\n\n` +
        `Chave PIX:\n${pixCode}\n\n` +
        `Ou escaneie o QR Code abaixo:`
      );
      await enviarImagemQRCode(numeroCliente, pixCode);
      break;
      
    case 'cartao':
      const linkPagamento = gerarLinkPagamentoCartao(pedidoId);
      await enviarMensagem(
        numeroCliente,
        `💳 *Pagamento via Cartão*\n\n` +
        `Clique no link para pagar:\n${linkPagamento}`
      );
      break;
      
    case 'boleto':
      const boletoUrl = gerarBoleto(pedidoId);
      await enviarDocumento(numeroCliente, boletoUrl, 'boleto.pdf');
      break;
  }
}
```

**Observações:** 
- WhatsApp não processa pagamentos
- Integrar com gateway de pagamento (Stripe, Mercado Pago, PagSeguro)
- Gerenciar carrinho no backend
- Confirmar pagamento antes de processar pedido

### 59. Links de produtos e coleções
**Descrição:** Compartilhar links diretos para produtos do catálogo.

**É possível implementar via API:** Sim

**Como implementar:**
- Gerar links wa.me com parâmetros de produto
- Ou enviar mensagens de produto específico via API

**Exemplo prático:**
```javascript
// Gerar link compartilhável de produto
function gerarLinkProduto(produtoId) {
  const mensagemPredefinida = encodeURIComponent(`Olá! Gostaria de saber mais sobre o produto ${produtoId}`);
  return `https://wa.me/${NUMERO_WHATSAPP_BUSINESS}?text=${mensagemPredefinida}`;
}

// Enviar produto específico (já mostrado anteriormente)
async function compartilharProduto(destinatario, produtoId) {
  await enviarProduto(destinatario, produtoId);
}

// QR Code para produto
async function gerarQRCodeProduto(produtoId) {
  const QRCode = require('qrcode');
  const link = gerarLinkProduto(produtoId);
  const qrCodeDataUrl = await QRCode.toDataURL(link);
  return qrCodeDataUrl;
}
```

**Observações:** Links wa.me funcionam para abrir conversa com mensagem pré-definida sobre produto.

**Como implementar:**
- Endpoint: `GET https://graph.facebook.com/v21.0/{phone_number_id}/analytics`
- Métricas disponíveis limitadas
- Complementar com analytics próprio

async function obterAnalyticsWhatsApp(inicio, fim) {
  const response = await fetch(
