class AudioRecorder {
  constructor() {
    this.mediaRecorder = null;
    this.audioChunks = [];
    this.stream = null;
    this.startTime = null;
    this.timerInterval = null;
  }
  
  async iniciar() {
    try {
      this.stream = await navigator.mediaDevices.getUserMedia({ 
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          sampleRate: 48000
        } 
      });
      
      // Usar formato compatível
      const mimeType = MediaRecorder.isTypeSupported('audio/webm;codecs=opus') 
        ? 'audio/webm;codecs=opus' 
        : 'audio/webm';
      
      this.mediaRecorder = new MediaRecorder(this.stream, {
        mimeType: mimeType
      });
      
      this.mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          this.audioChunks.push(event.data);
        }
      };
      
      this.mediaRecorder.onstop = async () => {
        const audioBlob = new Blob(this.audioChunks, { type: mimeType });
        await this.enviarAudio(audioBlob);
        this.limpar();
      };
      
      this.mediaRecorder.start();
      this.startTime = Date.now();
      this.iniciarTimer();
      
      // Mostrar UI de gravação
      document.getElementById('voice-recording-ui').style.display = 'flex';
      
    } catch (error) {
      console.error('Erro ao acessar microfone:', error);
      alert('Erro ao acessar microfone. Verifique as permissões.');
    }
  }
  
  parar() {
    if (this.mediaRecorder && this.mediaRecorder.state !== 'inactive') {
      this.mediaRecorder.stop();
      this.pararTimer();
    }
  }
  
  cancelar() {
    if (this.mediaRecorder) {
      this.mediaRecorder.stop();
      this.audioChunks = [];
    }
    this.limpar();
    document.getElementById('voice-recording-ui').style.display = 'none';
  }
  
  iniciarTimer() {
    const timerEl = document.querySelector('.recording-time');
    this.timerInterval = setInterval(() => {
      const elapsed = Math.floor((Date.now() - this.startTime) / 1000);
      const minutes = Math.floor(elapsed / 60);
      const seconds = elapsed % 60;
      timerEl.textContent = `${minutes}:${seconds.toString().padStart(2, '0')}`;
    }, 100);
  }
  
  pararTimer() {
    if (this.timerInterval) {
      clearInterval(this.timerInterval);
      this.timerInterval = null;
    }
  }
  
  async enviarAudio(audioBlob) {
    try {
      const formData = new FormData();
      formData.append('audio', audioBlob, 'voice-message.ogg');
      formData.append('destinatario', app.conversaAtual);
      formData.append('tipo', 'voice');
      
      const response = await fetch('/api/enviar-audio', {
        method: 'POST',
        body: formData
      });
      
      if (response.ok) {
        const mensagem = await response.json();
        app.adicionarMensagemNaUI({
          type: 'audio',
          isVoice: true,
          url: URL.createObjectURL(audioBlob),
          direcao: 'enviada',
          timestamp: new Date()
        });
      }
    } catch (error) {
      console.error('Erro ao enviar áudio:', error);
      alert('Erro ao enviar mensagem de voz');
    } finally {
      document.getElementById('voice-recording-ui').style.display = 'none';
    }
  }
  
  limpar() {
    if (this.stream) {
      this.stream.getTracks().forEach(track => track.stop());
      this.stream = null;
    }
    this.audioChunks = [];
    this.mediaRecorder = null;
    this.startTime = null;
  }
}

65. Mensagens temporárias (Disappearing Messages)
Descrição: Mensagens que se autodestroem após período configurado (24h, 7 dias, 90 dias).
É possível implementar via API: Não (oficialmente), Sim (localmente)
Como implementar:

API não suporta mensagens temporárias oficialmente
Implementar lógica de expiração no backend
Deletar mensagens automaticamente após período

Exemplo prático:
javascript// Estrutura de mensagem com expiração
const mensagem = {
  id: 'msg_123',
  texto: 'Esta mensagem expira em 24h',
  timestamp: new Date(),
  expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000), // 24h
  isTemporary: true
};

// Job para deletar mensagens expiradas
const cron = require('node-cron');

// Executar a cada hora
cron.schedule('0 * * * *', async () => {
  const agora = new Date();
  
  // Buscar mensagens expiradas
  const mensagensExpiradas = await db.mensagens.find({
    isTemporary: true,
    expiresAt: {$lte: agora},
    deleted: {$ne: true}
  });
  
  // Marcar como deletadas
  for (const msg of mensagensExpiradas) {
    await db.mensagens.update(
      {id: msg.id},
      {$set: {deleted: true, deletedAt: agora}}
    );
    
    // Notificar clientes conectados
    io.emit('message-deleted', {
      messageId: msg.id,
      conversaNumero: msg.from || msg.to
    });
  }
  
  console.log(`${mensagensExpiradas.length} mensagens temporárias deletadas`);
});

// Configurar mensagens temporárias para uma conversa
async function configurarMensagensTemporarias(numeroConversa, duracao) {
  // duracao: '24h', '7d', '90d', ou null para desativar
  
  let duracaoMs = null;
  if (duracao) {
    const match = duracao.match(/(\d+)([hd])/);
    const valor = parseInt(match[1]);
    const unidade = match[2];
    
    duracaoMs = unidade === 'h' 
      ? valor * 60 * 60 * 1000 
      : valor * 24 * 60 * 60 * 1000;
  }
  
  await db.conversas.update(
    {numero: numeroConversa},
    {
      $set: {
        mensagensTemporarias: duracao !== null,
        duracaoTemporaria: duracaoMs
      }
    }
  );
}

// Ao enviar mensagem, verificar se conversa tem temporárias ativadas
async function enviarMensagemComExpiracao(destinatario, texto) {
  const conversa = await db.conversas.findOne({numero: destinatario});
  
  const mensagem = {
    from: PHONE_NUMBER_ID,
    to: destinatario,
    texto: texto,
    timestamp: new Date(),
    isTemporary: conversa?.mensagensTemporarias || false
  };
  
  if (mensagem.isTemporary) {
    mensagem.expiresAt = new Date(Date.now() + conversa.duracaoTemporaria);
  }
  
  await db.mensagens.insert(mensagem);
  await enviarMensagem(destinatario, texto);
  
  return mensagem;
}

É possível implementar via API: Não (oficialmente), Sim (simulado)
Como implementar:

API não tem tipo "poll" nativo

Rastrear respostas no backend

javascript// Criar enquete
async function criarEnquete(destinatario, pergunta, opcoes) {
  const enqueteId = `poll_${Date.now()}`;

// Processar voto
async function processarVoto(numeroUsuario, buttonId) {
  const match = buttonId.match(/poll_(.+)_(\d+)/);
  if (!match) return;
  
  const enqueteId = match[1];
  const opcaoIndex = parseInt(match[2]);
  
  // Verificar se já votou
  const votoExistente = await db.votosEnquete.findOne({
    enqueteId: enqueteId,
    usuario: numeroUsuario
  });
  
  if (votoExistente) {
    await enviarMensagem(numeroUsuario, '❌ Você já votou nesta enquete!');
    return;
  }
  
  // Registrar voto
  await db.votosEnquete.insert({
    enqueteId: enqueteId,
    usuario: numeroUsuario,
    opcao: opcao```javascript
    opcao: opcaoIndex,
    timestamp: new Date()
  });
  
  // Atualizar contagem
  await db.enquetes.update(
    {id: enqueteId},
    {$inc: {[`opcoes.${opcaoIndex}.votos`]: 1}}
  );
  
  // Enviar confirmação com resultados parciais
  const enquete = await db.enquetes.findOne({id: enqueteId});
  const resultados = await gerarMensagemResultados(enquete);
  
  await enviarMensagem(numeroUsuario, `✅ Voto registrado!\n\n${resultados}`);
}

// Gerar mensagem com resultados
async function gerarMensagemResultados(enquete) {
  const totalVotos = enquete.opcoes.reduce((sum, op) => sum + op.votos, 0);
  
  let mensagem = `📊 *${enquete.pergunta}*\n\n`;
  mensagem += `Total de votos: ${totalVotos}\n\n`;
  
  enquete.opcoes.forEach((opcao, i) => {
    const percentual = totalVotos > 0 ? (opcao.votos / totalVotos * 100).toFixed(1) : 0;
    const barraCheia = Math.round(percentual / 10);
    const barraVazia = 10 - barraCheia;
    const barra = '█'.repeat(barraCheia) + '░'.repeat(barraVazia);
    
    mensagem += `${i + 1}. ${opcao.texto}\n`;
    mensagem += `   ${barra} ${percentual}% (${opcao.votos})\n\n`;
  });
  
  return mensagem;
}

### 67. Reações a mensagens (Reactions)
**Descrição:** Reagir a mensagens com emojis (curtir, amar, rir, etc).

**É possível implementar via API:** Sim (parcial)

**Como implementar:**
- API suporta envio de reações oficialmente
- Endpoint específico para reactions
- Receber reações via webhook

// Enviar reação a uma mensagem
async function reagirMensagem(messageId, emoji) {
  await fetch(`https://graph.facebook.com/v21.0/${PHONE_NUMBER_ID}/messages`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${ACCESS_TOKEN}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      messaging_product: "whatsapp",
      recipient_type: "individual",
      to: destinatario, // Número que enviou a mensagem original
      type: "reaction",
      reaction: {
        message_id: messageId,
        emoji: emoji // Ex: "👍", "❤️", "😂", "😮", "😢", "🙏"
      }
    })
  });
}

  if (message?.type === 'reaction') {
    const reaction = message.reaction;
    const messageId = reaction.message_id; // ID da mensagem que foi reagida
    const emoji = reaction.emoji; // Emoji da reação (vazio se removida)
    const from = message.from; // Quem reagiu

      await db.reacoes.insert({

        usuario: from,

        usuario: from
      });
    }
    
    // Atualizar UI em tempo real
    io.emit('reaction-update', {
      messageId: messageId,
      usuario: from,
      emoji: emoji
    });
  }
  
  res.sendStatus(200);
});

function renderizarReacoes(messageId, reacoes) {
  if (!reacoes || reacoes.length === 0) return '';
  
  // Agrupar por emoji

    contagemPorEmoji[r.emoji] = (contagemPorEmoji[r.emoji] || 0) + 1;

          ${emoji} ${count}
        </span>
      `).join('')}
    </div>
  `;
}

      messaging_product: "whatsapp",
      to: destinatario,
      type: "text",
      context: {
        message_id: messageIdOriginal
      },
      text: {
        body: texto
      }
    })
  });
}

app.post('/webhook', async (req, res) => {
  const message = req.body.entry?.[0]?.changes?.[0]?.value?.messages?.[0];
  
  if (message?.context) {
    const messageIdCitada = message.context.message_id;
    const from = message.context.from; // Quem enviou a mensagem citada
    
    // Buscar mensagem original
    const mensagemOriginal = await db.mensagens.findOne({
      messageId: messageIdCitada
    });
    
    // Salvar com referência
    await db.mensagens.insert({
      messageId: message.id,
      from: message.from,
      texto: message.text?.body,
      timestamp: new Date(message.timestamp * 1000),
      replyTo: messageIdCitada,
      replyToText: mensagemOriginal?.texto
    });
  }
  
  res.sendStatus(200);
});

class WhatsAppClone {
  // ... código anterior ...
  
  selecionarMensagemParaResponder(messageId, texto) {
    this.mensagemParaResponder = {id: messageId, texto: texto};
    
    // Mostrar preview no footer
    const footer = document.querySelector('.chat-footer');
    const previewExistente = footer.querySelector('.reply-preview-footer');
    
    if (previewExistente) {
      previewExistente.remove();
    }
    
    const preview = document.createElement('div');
    preview.className = 'reply-preview-footer';
    preview.innerHTML = `
      <div class="reply-bar"></div>
      <div class="reply-content">
        <span class="reply-label">Respondendo</span>
        <span class="reply-text">${texto.substring(0, 50)}${texto.length > 50 ? '...' : ''}</span>
      </div>
      <button class="close-reply" onclick="app.cancelarResposta()">
        <svg><!-- ícone X --></svg>
      </button>
    `;
    
    footer.insertBefore(preview, footer.firstChild);
    document.getElementById('message-input').focus();
  }
  
  cancelarResposta() {
    this.mensagemParaResponder = null;
    document.querySelector('.reply-preview-footer')?.remove();
  }
  
  async enviarMensagem() {
    const input = document.getElementById('message-input');
    const texto = input.value.trim();
    
    if (!texto || !this.conversaAtual) return;
    
    input.value = '';
    
    // Verificar se está respondendo
    if (this.mensagemParaResponder) {
      await responderMensagem(
        this.conversaAtual, 
        texto, 
        this.mensagemParaResponder.id
      );
      this.cancelarResposta();
    } else {
      await enviarMensagem(this.conversaAtual, texto);
    }
  }
}

  const messageText = messageEl.querySelector('.message-text').textContent;

  contextMenu.className = 'context-menu';
